cmake_minimum_required(VERSION 3.25)
project(waytoys_zones_bridge VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_AUTOMOC ON)
find_package(ECM REQUIRED NO_MODULE)
list(APPEND CMAKE_MODULE_PATH ${ECM_MODULE_PATH})
find_package(Qt6 REQUIRED COMPONENTS Core DBus Widgets)
find_package(KWin 6.3.6 EXACT REQUIRED CONFIG)
add_library(waytoys_zones_bridge MODULE bridge.cpp)
target_link_libraries(waytoys_zones_bridge PRIVATE KWin::kwin Qt6::DBus)
target_compile_options(waytoys_zones_bridge PRIVATE -Wall -Wextra -Werror)
set_target_properties(waytoys_zones_bridge PROPERTIES PREFIX "")
install(TARGETS waytoys_zones_bridge DESTINATION lib/x86_64-linux-gnu/qt6/plugins/kwin/plugins)
